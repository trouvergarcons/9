Option 1: Modern fetch() API

// 1. Send asynchronous GET request
const response = await fetch(apiUrl);

// 2. Parse JSON response body into JS object
const data = await response.json();

// 3. Update DOM with data properties
document.getElementById('cityName').textContent = 
  `${data.location.city}, ${data.location.country}`;
document.getElementById('tempValue').textContent = 
  data.current.temperature;

Option 2: Classic XMLHttpRequest

// 1. Create XMLHttpRequest instance
const xhr = new XMLHttpRequest();
xhr.open('GET', apiUrl, true);

// 2. Event listener triggers when response is ready
xhr.onload = function() {
  if (xhr.status >= 200 && xhr.status < 300) {
    // 3. Parse JSON string into JS object
    const data = JSON.parse(xhr.responseText);
    
    // 4. Update DOM
    document.getElementById('tempValue').textContent = 
      data.current.temperature;
  }
};
xhr.send();
